// debounce from script.ts
function debounce<T extends (...args: any[]) => void>(func: T, delay: number): (...args: Parameters<T>) => void {
    let timeout: ReturnType<typeof setTimeout>;
    return function(this: any, ...args: Parameters<T>): void {
        const context = this;
        clearTimeout(timeout);
        timeout = setTimeout(() => func.apply(context, args), delay);
    };
}

function contractorUrl(contractor: Contractor): string {

### Naming Conventions
- Components with JSX: PascalCase + .tsx (`ContractorDashboard.tsx`)
- Components without JSX: PascalCase + .jsx (`ContractorDashboard.jsx`)
- Hooks: camelCase with `use` prefix (`useContractor.ts`)
- Utils: camelCase (`formatDate.ts`)
- Types/Interfaces: PascalCase (`ContractorTypes.ts`)
- Constants: SCREAMING_SNAKE_CASE
- Files: kebab-case for non-component files
